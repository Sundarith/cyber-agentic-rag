% Main agentic reasoning path.

\node[widebox, fill=softblue, draw=agentblue] (planner) at (4.0,0)
  {Query Planner\\\subtext{route, decompose, choose tools}};
\node[widebox, fill=softblue, draw=agentblue] (controller) at (8.4,0)
  {Agent Controller\\\subtext{budgeted evidence loop}};
\node[widebox, fill=softviolet, draw=verifyviolet] (verifier) at (12.8,0)
  {Evidence Verifier\\\subtext{support, sufficiency, citations}};
\node[widebox, fill=softviolet, draw=verifyviolet] (synth) at (17.2,0)
  {Grounded Synthesizer\\\subtext{cited final response}};
\node[nodebox, fill=softred, draw=evalred] (answer) at (21.2,0)
  {Final Answer\\\subtext{auditable citations}};

\draw[mainflow] (query.east) -- (planner.west);
\draw[mainflow] (planner.east) -- (controller.west);
\draw[mainflow] (controller.east) -- (verifier.west);

\draw[mainflow] (synth.east) -- (answer.west);

% Tool-using agent layer.
\node[toolbox] (search) at (4.4,-2.2) {search\\hybrid};
\node[toolbox] (open) at (7.5,-2.2) {open\\exact ID};
\node[toolbox] (quote) at (10.6,-2.2) {find / quote\\section};
\node[toolbox] (summarize) at (13.7,-2.2) {summarize\\evidence};

\begin{pgfonlayer}{background}
\node[band, fit=(search)(open)(quote)(summarize)(compare)] (toolband) {};
\end{pgfonlayer}
\node[bandlabel] at ($(toolband.north west)+(2mm,-2mm)$) {Tool-Using Retrieval Agent};

% Single clean control/evidence path into and out of the tools.
\coordinate (tooltop) at ($(toolband.north)+(0,0)$);
\coordinate (toolbottom) at ($(toolband.south)+(0,0)$);
\draw[mainflow] (controller.south) -- node[right, font=\sffamily\scriptsize, text=agentblue] {tool calls} (tooltop);

% Retrieval, knowledge, graph, and memory substrate.
\node[storebox, draw=retrievalgreen] (hybrid) at (4.6,-4.9)
  {Hybrid Retrieval\\\subtext{BM25 + dense vectors + reranking}};
\node[storebox, draw=retrievalgreen] (hierarchy) at (9.2,-4.9)
  {Hierarchical Index\\\subtext{corpus $\rightarrow$ document $\rightarrow$ section $\rightarrow$ passage}};
\node[storebox, draw=graphorange] (graph) at (13.8,-4.9)
  {Graph Layer\\\subtext{entities, relations, claims, contradictions}};
\node[storebox, fill=softgray, draw=slate] (memory) at (18.4,-4.9)
  {Memory Layer\\\subtext{prior searches, summaries, failed paths}};

\begin{pgfonlayer}{background}
\node[band, fit=(hybrid)(hierarchy)(graph)(memory)] (knowledgeband) {};
\end{pgfonlayer}
\node[bandlabel] at ($(knowledgeband.north west)+(2mm,-2mm)$) {Retrieval, Knowledge, and Memory Substrate};

% Keep data-plane connections simple and readable.
\draw[flow] (toolbottom) -- node[right, font=\sffamily\scriptsize, text=slate] {retrieval actions} (knowledgeband.north);
\draw[flow] (hybrid.east) -- (hierarchy.west);
\draw[flow] (hierarchy.east) -- (graph.west);
\draw[flow] (graph.east) -- (memory.west);
\draw[flow] (knowledgeband.east) -- ++(7mm,0) |- node[pos=0.2, right, font=\sffamily\scriptsize, text=slate] {evidence bundle} (verifier.south);

\draw[feedback] (verifier.north) -- ++(0,8mm) -|
  node[pos=0.24, above, font=\sffamily\scriptsize, text=verifyviolet]

  (controller.north);

\node[nodebox, fill=softred, draw=evalred, text width=37mm] (trace) at (11.3,-7.6)

  {Evaluation Harness\\\subtext{golden, synthetic, adversarial, human review}};

\node[band, fit=(trace)(eval)] (evalband) {};

\node[bandlabel] at ($(evalband.north west)+(2mm,-2mm)$) {Measurement Layer};

\draw[flow] (trace.east) -- (eval.west);
\draw[traceflow] (controller.south east) -- ++(0,-9mm) -| (trace.north);
