    inner sep=4pt
  },
  smallbox/.style={

    nodebox,
    text width=42mm
  },
  band/.style={
    draw=linegray,
    rounded corners=4pt,

    inner sep=7pt
  },
  flow/.style={
    -{Latex[length=2.2mm]},
    draw=linegray,
    line width=0.5pt

\begin{tikzpicture}[node distance=8mm and 8mm]

% Main reasoning path.
\node[nodebox, fill=softblue, draw=agentblue] (query)
  {Query\\\subtext{CVE / CWE / CAPEC / ATT\&CK}};
\node[widebox, fill=softblue, draw=agentblue, right=of query] (planner)
  {Query Planner\\\subtext{route, decompose, choose tools}};
\node[widebox, fill=softblue, draw=agentblue, right=of planner] (controller)
  {Agent Controller\\\subtext{budgeted evidence loop}};
\node[widebox, fill=softviolet, draw=verifyviolet, right=of controller] (verifier)
  {Evidence Verifier\\\subtext{support, sufficiency, citations}};
\node[widebox, fill=softviolet, draw=verifyviolet, right=of verifier] (synth)
  {Grounded Synthesizer\\\subtext{cited final response}};
\node[nodebox, fill=softred, draw=evalred, right=of synth] (answer)
  {Final Answer\\\subtext{auditable citations}};

\draw[mainflow] (query) -- (planner);
\draw[mainflow] (planner) -- (controller);
\draw[mainflow] (controller) -- (verifier);
\draw[mainflow] (verifier) -- (synth);
\draw[mainflow] (synth) -- (answer);

% Tool layer.
\node[smallbox, fill=softteal, draw=toolteal, below=13mm of controller, xshift=-48mm] (search)
  {search\\hybrid};
\node[smallbox, fill=softteal, draw=toolteal, right=5mm of search] (open)
  {open\\exact ID};
\node[smallbox, fill=softteal, draw=toolteal, right=5mm of open] (quote)
  {find / quote\\section};
\node[smallbox, fill=softteal, draw=toolteal, right=5mm of quote] (summarize)
  {summarize\\evidence};
\node[smallbox, fill=softteal, draw=toolteal, right=5mm of summarize] (compare)
  {compare\\candidates};

\begin{pgfonlayer}{background}
\node[band, fit=(search)(open)(quote)(summarize)(compare)] (toolband) {};
\end{pgfonlayer}
\node[bandlabel] at ($(toolband.north west)+(2mm,-2mm)$) {Tool-Using Retrieval Agent};

\draw[mainflow] (controller.south) -- ++(0,-4mm) -| (open.north);
\draw[flow] (quote.north) -- ++(0,6mm) -| (verifier.south);

% Retrieval and knowledge substrate.
\node[widebox, fill=softgreen, draw=retrievalgreen, below=16mm of search, xshift=9mm] (hybrid)
  {Hybrid Retrieval\\\subtext{BM25 + dense vectors + reranking}};
\node[widebox, fill=softgreen, draw=retrievalgreen, right=7mm of hybrid] (hierarchy)
  {Hierarchical Index\\\subtext{corpus to document to section to passage}};

  {Graph Layer\\\subtext{entities, relations, claims, contradictions}};
\node[widebox, fill=softgray, draw=slate, right=7mm of graph] (memory)
  {Memory Layer\\\subtext{prior searches, summaries, failed paths}};

\begin{pgfonlayer}{background}
\node[band, fit=(hybrid)(hierarchy)(graph)(memory)] (knowledgeband) {};
\end{pgfonlayer}
\node[bandlabel] at ($(knowledgeband.north west)+(2mm,-2mm)$) {Retrieval, Knowledge, and Memory Substrate};

\draw[flow] (search.south) -- (hybrid.north);
\draw[flow] (open.south) -- (hierarchy.north);
\draw[flow] (quote.south) -- (hierarchy.north);
